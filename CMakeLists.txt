cmake_minimum_required(VERSION 3.22)
project(lingmo-wine VERSION 1.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
include(CTest)
find_package(LingmoSDK REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(WINE_HELPER "${CMAKE_INSTALL_FULL_LIBEXECDIR}/lingmo-wine/lingmo_wine.py" CACHE FILEPATH "Wine helper location")
add_executable(lingmo-wine src/main.cpp src/manager.cpp src/globalmenu.cpp)
target_link_libraries(lingmo-wine PRIVATE Lingmo::SDK)
target_compile_definitions(lingmo-wine PRIVATE WINE_HELPER="${WINE_HELPER}")
lingmo_add_translations(lingmo-wine SOURCES src)
install(TARGETS lingmo-wine RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES runtime/lingmo_wine.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/lingmo-wine)
install(FILES data/lingmo-mimeapps.list DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/lingmo-wine.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/lingmo-wine.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
if(BUILD_TESTING)
    add_executable(wine-window-test tests/window_test.cpp src/manager.cpp src/globalmenu.cpp)
    target_include_directories(wine-window-test PRIVATE src)
    target_link_libraries(wine-window-test PRIVATE Lingmo::SDK)
    target_compile_definitions(wine-window-test PRIVATE WINE_HELPER="${WINE_HELPER}")
    add_test(NAME wine-window COMMAND wine-window-test)
    add_executable(wine-management-test tests/management_test.cpp src/manager.cpp src/globalmenu.cpp)
    target_include_directories(wine-management-test PRIVATE src)
    target_link_libraries(wine-management-test PRIVATE Lingmo::SDK)
    target_compile_definitions(wine-management-test PRIVATE WINE_HELPER="${CMAKE_CURRENT_SOURCE_DIR}/runtime/lingmo_wine.py")
    add_test(NAME wine-management COMMAND wine-management-test)
    find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
    add_executable(wine-menu-test tests/menu_test.cpp src/globalmenu.cpp)
    target_include_directories(wine-menu-test PRIVATE src)
    target_link_libraries(wine-menu-test PRIVATE Lingmo::SDKBase)
    add_test(NAME wine-menu COMMAND ${DBUS_RUN_SESSION} -- $<TARGET_FILE:wine-menu-test>)
    add_test(NAME wine-runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_runtime.py)
    if(WINE_HELPER STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/runtime/lingmo_wine.py")
        set_tests_properties(wine-runtime PROPERTIES ENVIRONMENT "LINGMO_WINE_TEST_BINARY=$<TARGET_FILE:lingmo-wine>")
    endif()
endif()
